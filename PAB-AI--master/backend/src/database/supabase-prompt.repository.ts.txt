import { IPromptRepository } from './prompt.repository';
import { Prompt, PromptVersion } from '../models/prompt.model';
import { supabase } from '../config/supabase';

export class SupabasePromptRepository implements IPromptRepository {
  // ---------- Mappers ----------
  private mapDbToPrompt(row: any): Prompt {
    return {
      id: row.id,
      userId: row.user_id,
      name: row.name,
      content: row.content,
      description: row.description ?? undefined,
      category: row.category,
      isTemplate: row.is_template ?? false,
      version: row.version,
      createdAt: new Date(row.created_at),
      updatedAt: new Date(row.updated_at),
    };
  }

  private mapDbToVersion(row: any): PromptVersion {
    return {
      id: row.id,
      promptId: row.prompt_id,
      content: row.content,
      version: row.version,
      createdAt: new Date(row.created_at),
    };
  }

  // ---------- CRUD ----------
  async createPrompt(
    promptData: Omit<Prompt, 'id' | 'createdAt' | 'updatedAt' | 'version'>,
    userId: string,
  ): Promise<Prompt> {
    const { data, error } = await supabase
      .from('prompts')
      .insert([
        {
          user_id: userId,
          name: promptData.name,
          content: promptData.content,
          description: promptData.description,
          category: promptData.category,
          is_template: promptData.isTemplate,
          version: 1,
        },
      ])
      .select()
      .single();

    if (error) {
      throw new Error(`Supabase error creating prompt: ${error.message}`);
    }

    const prompt = this.mapDbToPrompt(data);

    // Insert initial version record
    const { error: vErr } = await supabase
      .from('prompt_versions')
      .insert([
        {
          prompt_id: prompt.id,
          content: prompt.content,
          version: 1,
        },
      ]);

    if (vErr) {
      throw new Error(`Supabase error creating prompt version: ${vErr.message}`);
    }

    return prompt;
  }

  async getUserPrompts(userId: string): Promise<Prompt[]> {
    const { data, error } = await supabase
      .from('prompts')
      .select('*')
      .eq('user_id', userId);

    if (error) {
      throw new Error(`Supabase error fetching user prompts: ${error.message}`);
    }

    return (data || []).map(this.mapDbToPrompt);
  }

  async getPromptById(promptId: string, userId: string): Promise<Prompt | null> {
    const { data, error } = await supabase
      .from('prompts')
      .select('*')
      .eq('id', promptId)
      .eq('user_id', userId)
      .maybeSingle();

    if (error) {
      throw new Error(`Supabase error fetching prompt by id: ${error.message}`);
    }

    return data ? this.mapDbToPrompt(data) : null;
  }

  async updatePrompt(
    promptId: string,
    userId: string,
    updates: Partial<Pick<Prompt, 'name' | 'content' | 'description' | 'category'>>,
  ): Promise<Prompt | null> {
    const existing = await this.getPromptById(promptId, userId);
    if (!existing) return null;

    const newVersion = existing.version + 1;
    const payload: any = {};

    if (updates.name !== undefined) payload.name = updates.name;
    if (updates.content !== undefined) payload.content = updates.content;
    if (updates.description !== undefined) payload.description = updates.description;
    if (updates.category !== undefined) payload.category = updates.category;

    payload.version = newVersion;
    payload.updated_at = new Date().toISOString();

    const { data, error } = await supabase
      .from('prompts')
      .update(payload)
      .eq('id', promptId)
      .eq('user_id', userId)
      .select()
      .single();

    if (error) {
      throw new Error(`Supabase error updating prompt: ${error.message}`);
    }

    const updatedPrompt = this.mapDbToPrompt(data);

    // Record new version if content changed
    if (updates.content !== undefined) {
      const { error: vErr } = await supabase
        .from('prompt_versions')
        .insert([
          {
            prompt_id: promptId,
            content: updates.content,
            version: newVersion,
          },
        ]);

      if (vErr) {
        throw new Error(`Supabase error creating prompt version: ${vErr.message}`);
      }
    }

    return updatedPrompt;
  }

  async deletePrompt(promptId: string, userId: string): Promise<boolean> {
    const { error } = await supabase
      .from('prompts')
      .delete()
      .eq('id', promptId)
      .eq('user_id', userId);

    if (error) {
      throw new Error(`Supabase error deleting prompt: ${error.message}`);
    }

    // Clean up associated versions
    await supabase.from('prompt_versions').delete().eq('prompt_id', promptId);
    return true;
  }

  async getPromptHistory(promptId: string): Promise<PromptVersion[]> {
    const { data, error } = await supabase
      .from('prompt_versions')
      .select('*')
      .eq('prompt_id', promptId)
      .order('version', { ascending: true });

    if (error) {
      throw new Error(`Supabase error fetching prompt history: ${error.message}`);
    }

    return (data || []).map(this.mapDbToVersion);
  }
}
